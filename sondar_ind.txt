import json, urllib.request, urllib.parse
B = "https://www.fnde.gov.br/olinda-ide/servico/DADOS_ABERTOS_SIOPE/versao/v1/odata"
for b in (3, 4):
    q = urllib.parse.urlencode({"@Ano_Consulta": "2026", "@Num_Peri": str(b), "@Sig_UF": "'AL'", "$format": "json",
                                "$top": "10000", "$filter": "COD_MUNI eq 270170"})
    u = f"{B}/Indicadores_Siope(Ano_Consulta=@Ano_Consulta,Num_Peri=@Num_Peri,Sig_UF=@Sig_UF)?{q}"
    with urllib.request.urlopen(urllib.request.Request(u, headers={"User-Agent": "painel-siope-fundeb"}), timeout=180) as f:
        v = [r for r in json.load(f)["value"] if str(r.get("COD_MUNI"))[:6] == "270170"]
    print(f"B{b} linhas={len(v)}")
    for r in v:
        print(" ", r.get("COD_EXIB"), "|", (r.get("NOM_INDI") or "")[:90], "|", r.get("VAL_INDI"))
